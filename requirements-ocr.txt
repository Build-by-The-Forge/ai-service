# OCR smoke-test runtime only
# Purpose: prove the current PP-OCRv3 baseline works on Python 3.12 Linux
#
# This file is the "core" layer: installed with normal dependency
# resolution, because these packages' own transitive deps are small,
# safe, and already proven to resolve cleanly (httpx, decorator, astor,
# opt-einsum for paddlepaddle).
#
# paddleocr==2.7.0 and its runtime deps live in requirements-ocr-nodeps.txt
# and are installed with --no-deps (see Dockerfile). Full resolution of
# paddleocr's declared deps pulls in PyMuPDF<1.21.0 (no Python 3.12 wheel,
# fails to build from source: needs Visual Studio on Windows / `swig` on
# Linux after ~18 min compiling MuPDF+Tesseract) and an unpinned
# `opencv-python` via imgaug, which silently overwrites the headless
# build below and reintroduces a libGL.so.1 dependency the slim image
# doesn't have. --no-deps plus exact pins avoids both.

paddlepaddle==2.6.2
# Deliberately above paddleocr 2.7's declared cap (<=4.6.0.66, not enforced
# under --no-deps): 4.6.0.66 bundles a vulnerable libwebp (PYSEC-2023-184).
# 4.11.0.86 is the newest release that accepts numpy 1.x, which
# paddlepaddle 2.6 needs; 4.12+ require numpy>=2. Real OCR output on the
# evaluation corpus is identical to 4.6.0.66 (OCR-stack security step).
opencv-python-headless==4.11.0.86
Pillow==12.3.0
# Native PDF text (services/ocr/pdf_parser.py). 1.28.0 has CPython 3.12
# abi3 wheels for Linux and Windows and no dependencies.
PyMuPDF==1.28.0
numpy==1.26.4
# 3.20.2 is affected by PYSEC-2026-1805 and PYSEC-2026-1806; 5.29.6 is the
# smallest upgrade that clears both, and switches to the compiled upb
# backend. paddlepaddle 2.6 accepts it on Linux (the image), where real OCR
# output on the evaluation corpus is identical to 3.20.2. On Windows it
# declares protobuf<=3.20.2, so local Windows installs keep that.
protobuf==5.29.6; sys_platform != "win32"
protobuf==3.20.2; sys_platform == "win32"
# paddle imports setuptools at load time, so it ships. 83.0.0 clears
# PYSEC-2025-49 and PYSEC-2026-3447.
setuptools==83.0.0
